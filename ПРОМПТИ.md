### 28. Повідомлення

~~~text
$ErrorActionPreference = 'Stop'\
\
\# Створюємо папку для результатів\
$folder = Join-Path $PSScriptRoot 'supplement'\
New-Item -ItemType Directory -Path $folder -Force | Out-Null\
\
$utf8 = New-Object System.Text.UTF8Encoding($false)\
$dnsResults = New-Object System.Collections.Generic.List[object]\
\
\# Виконуємо DNS-запит і зберігаємо відповідь\
function Save-DnsResult($fileName) {\
&#x20;   $startTime = [DateTimeOffset]::Now\.ToString('o')\
\
&#x20;   $result = @(Resolve-DnsName example.com -Server 1.1.1.1)\
\
&#x20;   $filePath = Join-Path $folder "$fileName.txt"\
&#x20;   $text = $result | Out-String -Width 4096\
\
&#x20;   [System.IO.File]::WriteAllText($filePath, $text, $utf8)\
\
&#x20;   \# Записуємо час запиту та отримані DNS-записи\
&#x20;   $queryInfo = [PSCustomObject]@{\
&#x20;       command = 'Resolve-DnsName example.com -Server 1.1.1.1'\
&#x20;       started = $startTime\
&#x20;       records = @($result | Select-Object Name, Type, IPAddress, TTL)\
&#x20;   }\
\
&#x20;   $dnsResults.Add($queryInfo)\
\
&#x20;   $json = $dnsResults | ConvertTo-Json -Depth 7\
&#x20;   $jsonPath = Join-Path $folder 'dns-metadata.json'\
\
&#x20;   [System.IO.File]::WriteAllText($jsonPath, $json, $utf8)\
}\
\
\# Починаємо відлік часу і робимо перший запит\
$timer = [System.Diagnostics.Stopwatch]::StartNew()\
Save-DnsResult 'dns-first'\
\
\# Беремо першу IPv4-адресу з відповіді\
$ip = $dnsResults[0].records |\
&#x20;   Where-Object { $\_.Type -eq 'A' } |\
&#x20;   Select-Object -First 1 -ExpandProperty IPAddress\
\
if (-not $ip) {\
&#x20;   throw 'У відповіді DNS не знайдено IPv4-адресу.'\
}\
\
\# Відкриваємо сайт через отриману IP-адресу\
$command = "curl.exe -v https\://example.com --resolve example.com:443:$ip"\
$curlFile = Join-Path $folder 'a1-resolve.txt'\
$startTime = [DateTimeOffset]::Now\.ToString('o')\
\
\# Зберігаємо відповідь curl разом із повідомленнями про з'єднання\
& $env:ComSpec /d /c "$command > \`"$curlFile\`" 2>&1"\
$exitCode = $LASTEXITCODE\
\
$curlInfo = [PSCustomObject]@{\
&#x20;   command   = $command\
&#x20;   started   = $startTime\
&#x20;   exit_code = $exitCode\
}\
\
$json = $curlInfo | ConvertTo-Json\
$jsonPath = Join-Path $folder 'curl-metadata.json'\
\
[System.IO.File]::WriteAllText($jsonPath, $json, $utf8)\
\
Write-Output 'Результат curl збережено.'\
\
\# Чекаємо, поки мине 6 хвилин від початку першого DNS-запиту\
while ($timer.Elapsed.TotalSeconds -lt 360) {\
&#x20;   Start-Sleep -Seconds 5\
}\
\
\# Повторюємо DNS-запит для порівняння результатів\
Save-DnsResult 'dns-second'\
$timer.Stop()\
\
Write-Output 'Другий DNS-запит збережено.'
~~~
